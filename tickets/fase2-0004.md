```python
class UserProfile(models.Model):
    user = models.OneToOneField(User, on_delete=models.CASCADE, related_name='profile')
    
    # Información pública
    bio = models.TextField(max_length=500, blank=True)
    location = models.CharField(max_length=100, blank=True)
    birth_date = models.DateField(null=True, blank=True)
    
    # Gaming preferences
    favorite_games = models.JSONField(default=list, blank=True)  # ["Magic", "Pokemon"]
    play_style = models.CharField(max_length=50, blank=True)  # "Competitive", "Casual", "Collector"
    experience_level = models.CharField(max_length=20, blank=True)  # "Beginner", "Intermediate", "Expert"
    
    # Personalización
    avatar_url = models.URLField(blank=True, null=True)
    banner_url = models.URLField(blank=True, null=True)
    theme_preference = models.CharField(max_length=20, default='system')  # 'light', 'dark', 'system'
    
    # Configuraciones de privacidad
    show_email = models.BooleanField(default=False)
    show_location = models.BooleanField(default=True)
    show_birth_date = models.BooleanField(default=False)
    show_communities = models.BooleanField(default=True)
    show_activity_stats = models.BooleanField(default=True)
    
    # Estadísticas (calculadas/cached)
    communities_count = models.IntegerField(default=0)
    posts_count = models.IntegerField(default=0)
    likes_received = models.IntegerField(default=0)
    reputation_score = models.IntegerField(default=0)  # Para Fase 4
    
    # Metadatos
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
```

**Serializers con privacidad:**
```python
class PublicUserProfileSerializer(serializers.ModelSerializer):
    profile = serializers.SerializerMethodField()
    communities = serializers.SerializerMethodField()
    
    def get_profile(self, obj):
        profile = obj.profile
        data = {
            'bio': profile.bio,
            'avatar_url': profile.avatar_url,
            'favorite_games': profile.favorite_games,
            'play_style': profile.play_style,
            'experience_level': profile.experience_level,
            'joined_at': obj.date_joined,
        }
        
        # Campos condicionales según privacidad
        if profile.show_location:
            data['location'] = profile.location
            
        if profile.show_activity_stats:
            data['stats'] = {
                'communities_count': profile.communities_count,
                'posts_count': profile.posts_count,
                'likes_received': profile.likes_received,
            }
            
        return data
```
