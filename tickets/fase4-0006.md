class RatingRateLimit:
    """
    Sistema de rate limiting para valoraciones
    """
    
    # Configuración por defecto (puede ser dinámica)
    # ✅ ACTUALIZADO: Enfoque UX-friendly con límites generosos
    LIMITS = {
        'daily_ratings': 15,     # Máximo 15 valoraciones por día (generoso)
        'weekly_ratings': 50,    # Máximo 50 valoraciones por semana (muy generoso)
        'same_user_cooldown': 5, # ✅ 5 días entre valoraciones al mismo usuario
        'new_user_daily': 5,     # Usuarios nuevos: 5 valoraciones por día
        'min_interactions': 3,   # ✅ NUEVO: Mínimo 3 interacciones reales
    }
    
    @staticmethod
    def can_rate_user(rater, rated_user):
        """Verificar si el usuario puede valorar"""
        
        # 1. Verificar límites diarios
        today_ratings = UserRating.objects.filter(
            rater=rater,
            created_at__date=timezone.now().date(),
            is_active=True
        ).count()
        
        daily_limit = RatingRateLimit.get_daily_limit(rater)
        if today_ratings >= daily_limit:
            return False, f"Límite diario alcanzado ({daily_limit})"
        
        # 2. Verificar límites semanales
        week_ago = timezone.now() - timedelta(days=7)
        week_ratings = UserRating.objects.filter(
            rater=rater,
            created_at__gte=week_ago,
            is_active=True
        ).count()
        
        if week_ratings >= RatingRateLimit.LIMITS['weekly_ratings']:
            return False, "Límite semanal alcanzado"
        
        # 3. ✅ NUEVO: Verificar interacciones reales mínimas
        interactions_count = RatingRateLimit.count_real_interactions(rater, rated_user)
        min_interactions = RatingRateLimit.LIMITS['min_interactions']
        if interactions_count < min_interactions:
            return False, f"Necesitas {min_interactions - interactions_count} interacciones más"
        
        # 4. Verificar cooldown con mismo usuario (✅ 5 días)
        last_rating = UserRating.objects.filter(
            rater=rater,
            rated_user=rated_user,
            is_active=True
        ).first()
        
        if last_rating:
            cooldown_days = RatingRateLimit.LIMITS['same_user_cooldown']  # 5 días
            if (timezone.now() - last_rating.created_at).days < cooldown_days:
                return False, f"Debes esperar {cooldown_days} días para valorar nuevamente"
        
        return True, "OK"
    
    @staticmethod
    def get_daily_limit(user):
        """Obtener límite diario basado en el perfil del usuario"""
        # Usuarios nuevos tienen límites más generosos (5 valoraciones)
        if user.date_joined > timezone.now() - timedelta(days=30):
            return RatingRateLimit.LIMITS['new_user_daily']  # 5 por día
        
        # Usuarios con baja reputación mantienen límite base
        if user.reputation_score < 2.0:
            return max(8, RatingRateLimit.LIMITS['daily_ratings'] // 2)  # Reducir a la mitad
        
        return RatingRateLimit.LIMITS['daily_ratings']  # 15 por día
    
    @staticmethod
    def count_real_interactions(user1, user2):
        """✅ NUEVO: Contar interacciones reales verificables entre usuarios"""
        from communities.models import Community, CommunityMembership
        from posts.models import Post, Comment
        from reactions.models import Reaction
        
        count = 0
        
        # 1. ¿Están en comunidades compartidas?
        shared_communities = Community.objects.filter(
            members__user__in=[user1, user2],
            members__status='active'
        ).distinct().count()
        
        if shared_communities == 0:
            return 0  # Sin comunidades compartidas = 0 interacciones
        
        # 2. Posts donde uno comentó al otro
        posts_interactions = Post.objects.filter(
            Q(author=user1, comments__author=user2) |
            Q(author=user2, comments__author=user1),
            is_active=True
        ).distinct().count()
        count += posts_interactions
        
        # 3. Comentarios donde uno respondió al otro  
        comment_replies = Comment.objects.filter(
            Q(author=user1, parent__author=user2) |
            Q(author=user2, parent__author=user1),
            is_active=True
        ).count()
        count += comment_replies
        
        # 4. Intercambio de reacciones (múltiples)
        reaction_exchanges = Reaction.objects.filter(
            Q(user=user1, content_object__in=get_user_content(user2)) |
            Q(user=user2, content_object__in=get_user_content(user1))
        ).count()
        count += min(reaction_exchanges // 3, 5)  # Max 5 puntos por reacciones
        
        return count

class InteractionValidator:
    """
    ✅ NUEVO: Sistema para validar interacciones reales entre usuarios
    Enfoque: Solo interacciones sociales (sin trading por ahora)
    """
    
    VALID_CONTEXTS = [
        'shared_community_membership',     # Miembros activos de misma comunidad
        'post_comment_exchanges',         # Comentarios en posts del otro usuario
        'comment_thread_conversations',   # Conversaciones en threads de comentarios
        'multiple_reaction_exchanges',    # Intercambio frecuente de reacciones
    ]
    
    @staticmethod
    def validate_rating_eligibility(rater, rated_user):
        """
        Validar si el usuario puede calificar basado en interacciones REALES
        """
        
        # 1. ¿No es auto-calificación?
        if rater == rated_user:
            return False, "No puedes calificarte a ti mismo"
        
        # 2. ¿Están en al menos UNA comunidad común activa?
        shared_communities = Community.objects.filter(
            members__user__in=[rater, rated_user],
            members__status='active'
        ).distinct().count()
        
        if shared_communities == 0:
            return False, "Deben pertenecer a una comunidad común"
        
        # 3. ¿Han tenido suficientes interacciones?
        interactions = RatingRateLimit.count_real_interactions(rater, rated_user)
        min_required = RatingRateLimit.LIMITS['min_interactions']  # 3
        
        if interactions < min_required:
            return False, f"Necesitas {min_required - interactions} interacciones más"
        
        return True, "Elegible para calificar"
    
    @staticmethod
    def get_interaction_history(user1, user2):
        """Obtener historial detallado de interacciones para mostrar al usuario"""
        
        history = {
            'shared_communities': [],
            'post_interactions': [],
            'comment_interactions': [],
            'reaction_count': 0,
            'total_score': 0
        }
        
        # Comunidades compartidas
        shared_communities = Community.objects.filter(
            members__user__in=[user1, user2],
            members__status='active'
        ).distinct()
        
        for community in shared_communities:
            history['shared_communities'].append({
                'name': community.name,
                'game_type': community.game_type.name if community.game_type else 'N/A'
            })
        
        # Posts con interacciones
        post_interactions = Post.objects.filter(
            Q(author=user1, comments__author=user2) |
            Q(author=user2, comments__author=user1),
            is_active=True
        ).distinct()[:5]  # Mostrar últimas 5
        
        for post in post_interactions:
            history['post_interactions'].append({
                'title': post.content[:50] + '...',
                'community': post.community.name,
                'date': post.created_at
            })
        
        # Calcular score total
        history['total_score'] = RatingRateLimit.count_real_interactions(user1, user2)
        
        return history

### Detección de Gaming y Patrones Sospechosos

```python
class AntiGamingDetector:
    """
    Sistema para detectar patrones de gaming y manipulación
    """
    
    @staticmethod
    def analyze_user_rating_patterns(user):
        """Analizar patrones sospechosos en valoraciones"""
        
        flags = []
        
        # 1. Detectar círculos de valoración mutua
        mutual_ratings = AntiGamingDetector.detect_mutual_rating_circles(user)
        if mutual_ratings['circle_size'] > 3:
            flags.append({
                'type': 'mutual_rating_circle',
                'severity': 'high',
                'details': mutual_ratings
            })
        
        # 2. Detectar valoraciones solo de 5 estrellas (sospechoso)
        five_star_percentage = AntiGamingDetector.calculate_five_star_bias(user)
        if five_star_percentage > 95 and user.ratings_given.count() > 10:
            flags.append({
                'type': 'five_star_bias',
                'severity': 'medium',
                'details': {'percentage': five_star_percentage}
            })
        
        # 3. Detectar valoraciones en ráfagas temporales
        burst_pattern = AntiGamingDetector.detect_rating_bursts(user)
        if burst_pattern['max_burst'] > 5:
            flags.append({
                'type': 'rating_burst',
                'severity': 'medium',
                'details': burst_pattern
            })
        
        # 4. Detectar valoraciones a usuarios inactivos
        inactive_ratings = AntiGamingDetector.detect_inactive_user_ratings(user)
        if inactive_ratings > 0.3:  # >30% a usuarios inactivos
            flags.append({
                'type': 'inactive_user_ratings',
                'severity': 'high',
                'details': {'percentage': inactive_ratings}
            })
        
        return flags
    
    @staticmethod
    def detect_mutual_rating_circles(user):
        """Detectar círculos de valoración mutua"""
        
        # Obtener usuarios que he valorado y que me han valorado
        rated_by_me = set(
            UserRating.objects.filter(rater=user, is_active=True)
            .values_list('rated_user_id', flat=True)
        )
        
        rated_me = set(
            UserRating.objects.filter(rated_user=user, is_active=True)
            .values_list('rater_id', flat=True)
        )
        
        # Intersección = valoración mutua
        mutual_ratings = rated_by_me.intersection(rated_me)
        
        return {
            'circle_size': len(mutual_ratings),
            'mutual_users': list(mutual_ratings),
            'percentage_mutual': len(mutual_ratings) / max(len(rated_by_me), 1) * 100
        }
    
    @staticmethod
    def detect_rating_bursts(user):
        """Detectar ráfagas de valoraciones en poco tiempo"""
        
        ratings = user.ratings_given.filter(is_active=True).order_by('created_at')
        
        bursts = []
        current_burst = []
        
        for rating in ratings:
            if not current_burst:
                current_burst = [rating]
            else:
                time_diff = rating.created_at - current_burst[-1].created_at
                if time_diff.total_seconds() < 3600:  # 1 hora
                    current_burst.append(rating)
                else:
                    if len(current_burst) > 1:
                        bursts.append(current_burst)
                    current_burst = [rating]
        
        # Agregar último burst si existe
        if len(current_burst) > 1:
            bursts.append(current_burst)
        
        max_burst = max([len(burst) for burst in bursts], default=0)
        
        return {
            'max_burst': max_burst,
            'total_bursts': len(bursts),
            'burst_details': [
                {
                    'size': len(burst),
                    'timespan': (burst[-1].created_at - burst[0].created_at).total_seconds() / 60
                }
                for burst in bursts
            ]
        }
```

```python
class AntiAbuseConfig(models.Model):
    """
    Configuración dinámica para sistema anti-abuso
    """
    key = models.CharField(max_length=100, unique=True)
    value = models.JSONField()
    description = models.TextField()
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    
    @classmethod
    def get_config(cls, key, default=None):
        """Obtener valor de configuración"""
        try:
            return cls.objects.get(key=key).value
        except cls.DoesNotExist:
            return default
    
    @classmethod
    def update_config(cls, key, value, description=""):
        """Actualizar configuración"""
        config, created = cls.objects.get_or_create(
            key=key,
            defaults={'value': value, 'description': description}
        )
        if not created:
            config.value = value
            config.description = description
            config.save()
```

## Comentarios y Discusión

### Filosofía Anti-Abuso

- **Prevención > Corrección**: Mejor prevenir que limpiar después
- **Transparencia Limitada**: No revelar todos los algoritmos para evitar evasión
- **Acciones Graduales**: Warning → Suspensión temporal → Ban permanente

### Métricas de Éxito

- **Reducción de flags**: <5% de valoraciones flagged por mes
- **Tiempo de resolución**: <24h para revisar flags críticos
- **Falsos positivos**: <2% de flags resultan en acciones incorrectas

### Balance Usabilidad-Seguridad

- **Rate limits razonables**: No frustrar usuarios legítimos
- **Feedback claro**: Explicar por qué se bloquea una acción
- **Proceso de apelación**: Permitir disputar suspensiones injustas
