## Descripción

Implementar un sistema completo para que los usuarios puedan ver y gestionar el historial de valoraciones que han recibido de otros usuarios. Esta funcionalidad permite transparencia en el sistema de reputación y ayuda a los usuarios a entender cómo otros perciben sus interacciones en la comunidad TCG.

El historial debe ser informativo, bien organizado y respetar la privacidad, mostrando valoraciones de forma constructiva para fomentar mejores interacciones.

```python
class RatingHistoryFilter(django_filters.FilterSet):
    """Filtros para historial de valoraciones"""
    
    rating = django_filters.NumberFilter()
    rating_min = django_filters.NumberFilter(field_name='rating', lookup_expr='gte')

    date_from = django_filters.DateFilter(field_name='created_at', lookup_expr='gte')
    date_to = django_filters.DateFilter(field_name='created_at', lookup_expr='lte')
    
    interaction_type = django_filters.ChoiceFilter(choices=INTERACTION_TYPE_CHOICES)
    
    has_comment = django_filters.BooleanFilter(method='filter_has_comment')
    
    def filter_has_comment(self, queryset, name, value):
        if value:
            return queryset.exclude(comment='')
        return queryset.filter(comment='')

        model = UserRating
```

```python
class RatingHistorySerializer(serializers.ModelSerializer):
    """Serializer para historial de valoraciones recibidas"""
    
    rater_info = serializers.SerializerMethodField()
    time_ago = serializers.SerializerMethodField()
    can_respond = serializers.SerializerMethodField()
    response = serializers.SerializerMethodField()
    
    class Meta:
        model = UserRating
        fields = [
            'id', 'rating', 'comment', 'interaction_type',
            'created_at', 'time_ago', 'rater_info', 
            'can_respond', 'response'
        ]
    
    def get_rater_info(self, obj):
        """Información del evaluador (puede ser anónima)"""
        user = self.context['request'].user
        
        # Check privacy settings
        if user.profile.hide_rater_names:
            return {
                'username': 'Usuario Anónimo',
                'avatar_url': None,
                'reputation_score': obj.rater.reputation_score
            }
        
        return {
            'id': obj.rater.id,
            'username': obj.rater.username,
            'avatar_url': obj.rater.avatar_url,
            'reputation_score': obj.rater.reputation_score
        }
    
    def get_time_ago(self, obj):
        return naturaltime(obj.created_at)
    
    def get_can_respond(self, obj):
        # Solo puedo responder si no he respondido y rating tiene comentario
        return bool(obj.comment and not hasattr(obj, 'response'))
    
    def get_response(self, obj):
        if hasattr(obj, 'response'):
            return {
                'content': obj.response.content,
                'created_at': obj.response.created_at
            }
        return None
```

```python
class RatingResponse(models.Model):
    """
    Respuestas a comentarios de valoraciones
    """
    rating = models.OneToOneField(UserRating, on_delete=models.CASCADE, related_name='response')
    content = models.TextField(max_length=300)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    is_active = models.BooleanField(default=True)
    
    class Meta:
        ordering = ['-created_at']

class RatingReport(models.Model):
    """
    Reportes de valoraciones inapropiadas
    """
    rating = models.ForeignKey(UserRating, on_delete=models.CASCADE)
    reporter = models.ForeignKey(User, on_delete=models.CASCADE)
    reason = models.CharField(max_length=50, choices=REPORT_REASON_CHOICES)
    description = models.TextField(max_length=500)
    created_at = models.DateTimeField(auto_now_add=True)
    status = models.CharField(max_length=20, default='pending')
    
    class Meta:
        unique_together = ['rating', 'reporter']
```
